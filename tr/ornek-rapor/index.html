04_ACCEPTANCE_TESTS.md (Deterministik Test Paketi)

import { test, expect } from '@playwright/test';

test('llms.txt uç noktası 200 OK ve text/plain döndürmeli', async ({ request }) => {
  const response = await request.get('https://ornek-e-ticaret-sitesi.example/llms.txt');
  expect(response.status()).toBe(200);
  expect(response.headers()['content-type']).toContain('text/markdown');
});

test('robots.txt PerplexityBot ve OAI-SearchBot izinlerini içermeli', async ({ request }) => {
  const response = await request.get('https://ornek-e-ticaret-sitesi.example/robots.txt');
  const body = await response.text();
  expect(body).not.toContain('User-agent: PerplexityBot
Disallow: /');
  expect(body).not.toContain('User-agent: OAI-SearchBot
Disallow: /');
});
